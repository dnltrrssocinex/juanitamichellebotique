<script>(() => {
  const root = document.getElementById('jm-nav-widget');
  if (!root || root.dataset.ready) return;
  root.dataset.ready = 'true';
  const config = JSON.parse(document.getElementById('jm-navbar-settings').textContent);
  const home = new URL(config.homeUrl);
  const path = value => value.replace(/\/+$/, '') || '/';
  const onHomepage = () => location.origin === home.origin && path(location.pathname) === path(home.pathname);
  const localPreview = () => ['localhost','127.0.0.1'].includes(location.hostname);
  const header = root.querySelector('.jmn-header');
  const spacer = document.getElementById('jm-nav-space');
  const nav = root.querySelector('#jm-navigation');
  const menu = root.querySelector('.jmn-menu');
  const quiz = root.querySelector('#jm-style-dialog');
  const booking = root.querySelector('#jm-booking-dialog');
  const content = root.querySelector('#jm-quiz-content');
  const form = root.querySelector('#jm-reservation-form');
  const status = root.querySelector('#jm-form-status');
  const success = root.querySelector('#jm-booking-success');
  let step = 0, answers = [], quizResult = null, sending = false;
  let previousOverflow = '';
  // Escape GHL ancestors that apply transforms or clipping to Custom Code elements.
  document.body.appendChild(root);
  const resize = new ResizeObserver(() => { spacer.style.height = header.offsetHeight + 'px'; });
  resize.observe(header);
  spacer.style.height = header.offsetHeight + 'px';
  function closeMenu() {
    nav.classList.remove('is-open'); menu.setAttribute('aria-expanded', 'false');
    menu.setAttribute('aria-label', 'Open navigation menu'); menu.textContent = '☰';
  }
  menu.addEventListener('click', () => {
    const open = nav.classList.toggle('is-open');
    menu.setAttribute('aria-expanded', String(open));
    menu.setAttribute('aria-label', open ? 'Close navigation menu' : 'Open navigation menu');
    menu.textContent = open ? '×' : '☰';
  });
  const headings = {experiences:'More Than Shopping — An Experience',about:'Meet Juanita Michelle',visit:'Visit & Contact',contact:'Visit & Contact'};
  function section(key) {
    // Never mistake a shop footer anchor for a homepage section.
    if (!onHomepage() && !localPreview()) return null;
    // IDs are editable for native GHL sections; heading lookup also supports the exported homepage.
    let target;
    try { target = document.querySelector(config.sections[key]); } catch { return null; }
    if (!target) target = [...document.querySelectorAll('h2')].find(h => h.textContent.trim() === headings[key])?.closest('section');
    return target;
  }
  function jump(key, updateHash = true) {
    const target = section(key);
    if (!target) return false;
    closeMenu();
    const top = target.getBoundingClientRect().top + window.scrollY - header.offsetHeight - 16;
    window.scrollTo({top: Math.max(0, top), behavior: matchMedia('(prefers-reduced-motion: reduce)').matches ? 'instant' : 'smooth'});
    if (updateHash) history.replaceState(null, '', '#' + key);
    nav.querySelectorAll('[data-section]').forEach(link => {
      if (link.dataset.section === key) link.setAttribute('aria-current', 'location');
      else link.removeAttribute('aria-current');
    });
    return true;
  }
  const normalClick = event => !event.ctrlKey && !event.metaKey && !event.shiftKey && !event.altKey && event.button === 0;
  function navigate(event, url) {
    if (!normalClick(event)) return;
    event.preventDefault(); event.stopPropagation(); closeMenu();
    window.location.assign(url);
  }
  root.querySelectorAll('[data-home]').forEach(link => {
    link.href = home.href;
    link.addEventListener('click', event => {
      if (!normalClick(event)) return;
      if (onHomepage()) {
        event.preventDefault();event.stopPropagation();closeMenu();
        history.replaceState(null,'',home.pathname + home.search);
        window.scrollTo({top:0,behavior:'smooth'});
        nav.querySelectorAll('[aria-current]').forEach(item=>item.removeAttribute('aria-current'));
        nav.querySelector('[data-home]')?.setAttribute('aria-current','page');
      } else navigate(event,home.href);
    },true);
  });
  root.querySelectorAll('[data-shop]').forEach(link => {
    link.href = config.shopUrl;
    link.addEventListener('click', event => navigate(event,link.href),true);
  });
  if (onHomepage() && !location.hash) nav.querySelector('[data-home]')?.setAttribute('aria-current','page');
  if (path(location.pathname) === path(new URL(config.shopUrl).pathname)) nav.querySelector('[data-shop]')?.setAttribute('aria-current','page');
  root.querySelectorAll('[data-section]').forEach(link => {
    link.href = home.href.split('#')[0] + '#' + link.dataset.section;
    link.addEventListener('click', event => {
      if (!normalClick(event)) return;
      event.preventDefault(); event.stopPropagation();
      nav.querySelectorAll('[data-home],[data-shop]').forEach(item=>item.removeAttribute('aria-current'));
      if (!jump(link.dataset.section)) window.location.assign(link.href);
    },true);
  });
  // Revisit hash targets after the homepage renderer and images finish loading.
  const followHash = () => {
    const key = location.hash.slice(1);
    if (Object.hasOwn(config.sections, key)) return jump(key, false);
    return false;
  };
  let hashFrame = 0;
  const queueHash = () => {
    cancelAnimationFrame(hashFrame);
    hashFrame = requestAnimationFrame(() => requestAnimationFrame(followHash));
  };
  // GHL and the homepage renderer can insert content after the initial page load.
  // Keep watching briefly so their initial scroll resets do not override the destination.
  const sectionObserver = new MutationObserver(queueHash);
  sectionObserver.observe(document.body, {childList:true,subtree:true});
  queueHash();
  setTimeout(() => sectionObserver.disconnect(), 10000);
  window.addEventListener('hashchange', followHash);
  window.addEventListener('load', queueHash, {once:true});

  function openDialog(dialog) {
    closeMenu();
    if (!quiz.open && !booking.open) { previousOverflow = document.body.style.overflow; document.body.style.overflow = 'hidden'; }
    dialog.showModal();
  }
  [quiz, booking].forEach(dialog => {
    dialog.querySelectorAll('[data-close]').forEach(button => button.addEventListener('click', () => dialog.close()));
    dialog.addEventListener('click', event => {
      if (event.target !== dialog) return;
      const r = dialog.getBoundingClientRect();
      if (event.clientX < r.left || event.clientX > r.right || event.clientY < r.top || event.clientY > r.bottom) dialog.close();
